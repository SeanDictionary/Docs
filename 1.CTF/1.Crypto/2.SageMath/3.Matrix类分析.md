# Matrix

SageMath 中 Matrix 对象在对于不同参数下会产生不同的子类实例。

## Matrix(Zmod(n))

在 $1 < n < 2^8$ 时，会使用 `sage.matrix.matrix_modn_dense_float.Matrix_modn_dense_float`

在 $2^8 \le n < 94906266$ 时，会使用 `sage.matrix.matrix_modn_dense_double.Matrix_modn_dense_double`

在 $n \ge 94906266$ 时，会使用 `sage.matrix.matrix_generic_dense.Matrix_generic_dense`

```python
cimport sage.matrix.matrix as matrix
from sage.matrix import matrix_dense


cdef class Matrix_dense(matrix.Matrix):
    ...
cdef class Matrix_generic_dense(matrix_dense.Matrix_dense):
    ...
cdef class Matrix_modn_dense_template(matrix_dense.Matrix_dense):
    ...
cdef class Matrix_modn_dense_float(Matrix_modn_dense_template):
    ...
cdef class Matrix_modn_dense_double(Matrix_modn_dense_template):
    ...
```

从源码看，这三个类都是继承自 `sage.matrix.matrix_dense.Matrix_dense`

### 随机数

`random_matrix()` 可以用来生成随机数矩阵，通过调用 `A.randomize()` 方法实现。

```python
# sage/matrix/special.py
@matrix_method
def random_matrix(ring, nrows, ncols=None, algorithm='randomize', implementation=None, *args, **kwds):
    if ncols is None:
        ncols = nrows
    sparse = kwds.pop('sparse', False)
    # Construct the parent of the desired matrix
    parent = matrix_space.MatrixSpace(ring, nrows, ncols, sparse=sparse, implementation=implementation)
    if algorithm == 'randomize':
        density = kwds.pop('density', None)
        # zero matrix is immutable, copy is mutable
        A = copy(parent.zero_matrix())
        if density is None:
            A.randomize(density=float(1), nonzero=False, *args, **kwds)
        else:
            A.randomize(density=density, nonzero=True, *args, **kwds)
        return A
    elif algorithm == 'echelon_form':
        return random_rref_matrix(parent, *args, **kwds)
    elif algorithm == 'echelonizable':
        return random_echelonizable_matrix(parent, *args, **kwds)
    elif algorithm == 'diagonalizable':
        return random_diagonalizable_matrix(parent, *args, **kwds)
    elif algorithm == 'subspaces':
        return random_subspaces_matrix(parent, *args, **kwds)
    elif algorithm == 'unimodular':
        return random_unimodular_matrix(parent, *args, **kwds)
    elif algorithm == 'unitary':
        return random_unitary_matrix(parent, *args, **kwds)
    else:
        raise ValueError('random matrix algorithm "%s" is not recognized' % algorithm)
```

在 `sage/matrix/matrix2.pyx` 中找到父类最初定义的 `randomize`：

```python
cdef class Matrix(Matrix1):
    ...
    def randomize(self, density=1, nonzero=False, *args, **kwds):
        randint = current_randstate().python_random().randint

        density = float(density)
        if density <= 0:
            return
        if density > 1:
            density = 1

        self.check_mutability()
        self.clear_cache()

        R = self.base_ring()

        cdef Py_ssize_t i, j, num

        if nonzero:
            if density >= 1:
                for i from 0 <= i < self._nrows:
                    for j from 0 <= j < self._ncols:
                        self.set_unsafe(i, j, R._random_nonzero_element(*args, **kwds))
            else:
                num = int(self._nrows * self._ncols * density)
                for i from 0 <= i < num:
                    self.set_unsafe(randint(0, self._nrows - 1),
                                    randint(0, self._ncols - 1),
                                    R._random_nonzero_element(*args, **kwds))
        else:
            if density >= 1:
                for i from 0 <= i < self._nrows:
                    for j from 0 <= j < self._ncols:
                        self.set_unsafe(i, j, R.random_element(*args, **kwds))
            else:
                num = int(self._nrows * self._ncols * density)
                for i from 0 <= i < num:
                    self.set_unsafe(randint(0, self._nrows - 1),
                                    randint(0, self._ncols - 1),
                                    R.random_element(*args, **kwds))
```

在 `sage/matrix/matrix_modn_dense_template.pxi` 对 `Matrix_modn_dense_template` 类重写了 `randomize` 方法：

```python
cdef class Matrix_modn_dense_template(Matrix_dense):
    ...
    def randomize(self, density=1, nonzero=False):
        density = float(density)
        if density <= 0:
            return
        if density > 1:
            density = float(1)

        self.check_mutability()
        self.clear_cache()

        cdef randstate rstate = current_randstate()

        cdef int nc, p = <int>self.p
        cdef long pm1

        if not nonzero:
            # Original code, before adding the ``nonzero`` option.
            if density == 1:
                for i from 0 <= i < self._nrows*self._ncols:
                    self._entries[i] = rstate.c_random() % p
            else:
                nc = self._ncols
                num_per_row = int(density * nc)
                sig_on()
                for i from 0 <= i < self._nrows:
                    for j from 0 <= j < num_per_row:
                        k = rstate.c_random() % nc
                        self._matrix[i][k] = rstate.c_random() % p
                sig_off()
        else:
            # New code, to implement the ``nonzero`` option.
            pm1 = p - 1
            if density == 1:
                for i from 0 <= i < self._nrows*self._ncols:
                    self._entries[i] = (rstate.c_random() % pm1) + 1
            else: 
                nc = self._ncols
                num_per_row = int(density * nc)
                sig_on()
                for i from 0 <= i < self._nrows:
                    for j from 0 <= j < num_per_row:
                        k = rstate.c_random() % nc
                        self._matrix[i][k] = (rstate.c_random() % pm1) + 1
                sig_off()
```

可以归纳出:

在 $n < 94906266$ 时，会使用 `c_random()` 生成随机数，最终可以定位到 `GMP` 的 MT19937 过程。

在 $n \ge 94906266$ 时，会使用 `R.random_element()` 生成随机数，最终可以定位到 `Python/random` 的 MT19937 过程。

详细的随机数生成过程可以在[4.rand.md](./4.rand.md)中查看。
