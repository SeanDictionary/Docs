# 随机数

SageMath 中主要有两类随机数生成器，一类是 Python 自带的 random 模块，另一类是 GMP 库中的随机数生成器。

可以使用 `set_random_seed()` 函数设置 GMP 随机数生成器种子，然后使用 GMP 计算 `ZZ.random_element(1 << 128)` 作为 random 随机数生成器种子（128bits）

## 两类随机数生成器

### 1.GMP

分析 `GMP` 的 MT19937 过程

```python
# sage/misc/randstate.pyx
cdef class randstate:
    ...
    cpdef int c_random(self) noexcept:
        return gmp_urandomb_ui(self.gmp_state, 31)
```

`c_random` 调用了 `gmp_urandomb_ui` 输出了 31bits 的随机数

```C
// GMP/rand/randbui.c
unsigned long gmp_urandomb_ui (gmp_randstate_ptr rstate, unsigned long bits)
{
  mp_limb_t  a[LIMBS_PER_ULONG];

  /* start with zeros, since if bits==0 then _gmp_rand will store nothing at
     all, or if bits <= GMP_NUMB_BITS then it will store only a[0] */
  a[0] = 0;
#if LIMBS_PER_ULONG > 1
  a[1] = 0;
#endif

  _gmp_rand (a, rstate, MIN (bits, BITS_PER_ULONG));

#if LIMBS_PER_ULONG == 1
  return a[0];
#else
  return a[0] | (a[1] << GMP_NUMB_BITS);
#endif
}
```

在 `gmp_urandomb_ui` 中调用了 `_gmp_rand` 函数，用于产生随机数

在 `sage/misc/randstate.pyx` 中的 `randstate` 类中定义了有关 `python_random` 的方法：

```python
import random as _random

DEFAULT_PYTHON_RANDOM = _random.Random

cpdef randstate current_randstate():
    return _current_randstate

cpdef set_random_seed(seed=None):
    global _current_randstate
    _current_randstate = randstate(seed)

cdef class randstate:
    def __init__(self, seed=None):
        cdef mpz_t mpz_seed

        if seed is None:
            if use_urandom:
                seed = int(binascii.hexlify(os.urandom(16)), 16)
            else:
                seed = int(time.time() * 256)
        else:
            seed = int(seed)

        # If seed==0, leave it at the default seed used by
        # gmp_randinit_default()
        if seed:
            mpz_init(mpz_seed)
            mpz_set_pylong(mpz_seed, seed)
            gmp_randseed(self.gmp_state, mpz_seed)
            mpz_clear(mpz_seed)

        self._seed = seed
    ...
    def python_random(self, cls=None, seed=None):
        if cls is None:
            cls = DEFAULT_PYTHON_RANDOM

        if type(self._python_random) is cls:
            return self._python_random

        from sage.rings.integer_ring import ZZ
        rand = cls()
        if seed is None:
            rand.seed(int(ZZ.random_element(1<<128)))
        else:
            rand.seed(int(seed))
        self._python_random = rand
        return rand
```

可以看到 `python_random` 方法会返回一个 `random.Random` 的实例，并且会使用 `ZZ.random_element(1<<128)` 作为 seed 来初始化该实例

另外查看源码 `ZZ.random_element()` 使用的是 GMP 作为随机数生成器，受 `set_random_seed()` 传入的 seed 控制

## 部分随机数生成函数

### _gmp_rand()

继续寻找能发现这个函数是一个通用的随机数输出函数

```C
/* GMP/gmp-impl.h */
typedef __gmp_randstate_struct *gmp_randstate_ptr;
typedef struct {
  void (*randseed_fn) (gmp_randstate_t, mpz_srcptr);
  void (*randget_fn) (gmp_randstate_t, mp_ptr, unsigned long int);
  void (*randclear_fn) (gmp_randstate_t);
  void (*randiset_fn) (gmp_randstate_ptr, gmp_randstate_srcptr);
} gmp_randfnptr_t;

#define RNG_FNPTR(rstate) ((rstate)->_mp_algdata._mp_lc)

#define _gmp_rand(rp, state, bits)					                \
  do {									                            \
    gmp_randstate_ptr  __rstate = (state);				            \
    (*((gmp_randfnptr_t *) RNG_FNPTR (__rstate))->randget_fn)		\
      (__rstate, rp, bits);						                    \
  } while (0)
```

可以从中推出 `RNG_FNPTR` 获取了 `gmp_randfnptr_t` 结构体指针，然后调用了该实例的 `randget_fn` 函数用于输出随机数

寻找在 GMP 中默认使用的 `gmp_randfnptr_t` 实例，发现 `randget_fn` 指向 ` __gmp_randget_mt`

```C
/* GMP/rand/randmts.c */
static const gmp_randfnptr_t Mersenne_Twister_Generator = {
  randseed_mt,
  __gmp_randget_mt,
  __gmp_randclear_mt,
  __gmp_randiset_mt
};
```

最终确定出 GMP 使用的是 MT19937，主要实现在 `GMP/rand/randmts.c` 和 `GMP/rand/randmt.c` 中

GMP 中 MT19937 的具体过程如下

1. 计算 $seed_1=seed\mod (2^{19937}-20027)+2$
2. 计算 $seed_2={seed_1}^{1074888996}\mod (2^{19937}-20023)$
3. 设置 $mt_0=\begin{cases}2^{32}&,seed_2>>19936=1\\0&,seed_2>>19936=0\end{cases}$
4. 状态数组其余位置通过 $mt_i=seed_2 >> (32\cdot(i-1))\mod 2^{32}$ 得到
5. 使用默认参数 `WARM_UP=2000` 预热随机数生成器

### ZZ.random_element()

```python
# sage/rings/integer_ring.pyx
cdef class IntegerRing_class(CommutativeRing):
    ...
    def random_element(self, x=None, y=None, distribution=None):
        cdef Integer z = Integer.__new__(Integer)
        if distribution == "1/n":
            x = None
            y = None
        elif distribution == "mpz_rrandomb" or distribution == "gaussian":
            y = None
        if x is not None and y is None and x <= 0:
            raise TypeError("x must be > 0")
        if x is not None and y is not None and x >= y:
            raise TypeError("x must be < y")
        self._randomize_mpz(z.value, x, y, distribution)
        return z

    cdef int _randomize_mpz(self, mpz_t value, x, y, distribution) except -1:
        cdef Integer r
        cdef Integer n_max, n_min, n_width
        cdef randstate rstate = current_randstate()
        cdef int den = rstate.c_random()-SAGE_RAND_MAX/2
        if den == 0:
            den = 1
        if (distribution is None and x is None) or distribution == "1/n":
            mpz_set_si(value, (SAGE_RAND_MAX/5*2) / den)
        elif distribution is None or distribution == "uniform":
            if y is None:
                if x is None:
                    mpz_set_si(value, rstate.c_random() % 5 - 2)
                else:
                    n_max = x if isinstance(x, Integer) else self(x)
                    mpz_urandomm(value, rstate.gmp_state, n_max.value)
            else:
                n_min = x if isinstance(x, Integer) else self(x)
                n_max = y if isinstance(y, Integer) else self(y)
                n_width = n_max - n_min
                if mpz_sgn(n_width.value) <= 0:
                    n_min = self(-2)
                    n_width = self(5)
                mpz_urandomm(value, rstate.gmp_state, n_width.value)
                mpz_add(value, value, n_min.value)
        elif distribution == "mpz_rrandomb":
            if x is None:
                raise ValueError("must specify x to use 'distribution=mpz_rrandomb'")
            mpz_rrandomb(value, rstate.gmp_state, int(x))
        elif distribution == "gaussian":
            global _prev_discrete_gaussian_integer_sampler
            if x == _prev_discrete_gaussian_integer_sampler[0]:
                r = _prev_discrete_gaussian_integer_sampler[1]()
            else:
                from sage.stats.distributions.discrete_gaussian_integer import DiscreteGaussianDistributionIntegerSampler
                D = DiscreteGaussianDistributionIntegerSampler(sigma=x, algorithm='uniform+logtable')
                r = D()
                _prev_discrete_gaussian_integer_sampler = (x, D)
            mpz_set(value, r.value)
        else:
            raise ValueError("Unknown distribution for the integers: %s" % distribution)
```

`random_element` 调用了 GMP 的 `mpz_urandomm` 来生成随机数

```C
/* GMP/mpz/urandomm.c */
#define MAX_URANDOMM_ITER  80

void
mpz_urandomm (mpz_ptr rop, gmp_randstate_t rstate, mpz_srcptr n)
{
  mp_ptr rp, np;
  mp_size_t nbits, size;
  mp_limb_t nh;
  int count;
  int pow2;
  int cmp;
  TMP_DECL;

  size = ABSIZ (n);
  if (UNLIKELY (size == 0))
    DIVIDE_BY_ZERO;

  np = PTR (n);
  nh = np[size - 1];

  /* Detect whether n is a power of 2.  */
  pow2 = POW2_P (nh) && (size == 1 || mpn_zero_p (np, size - 1));

  count_leading_zeros (count, nh);
  nbits = size * GMP_NUMB_BITS - (count - GMP_NAIL_BITS) - pow2;
  if (nbits == 0)		/* nbits == 0 means that n was == 1.  */
    { 
      SIZ (rop) = 0;
      return;
    }

  TMP_MARK;
  if (rop == n)
    {
      mp_ptr tp;
      tp = TMP_ALLOC_LIMBS (size);
      MPN_COPY (tp, np, size);
      np = tp;
    }

  /* Here the allocated size can be one too much if n is a power of
     (2^GMP_NUMB_BITS) but it's convenient for using mpn_cmp below.  */
  rp = MPZ_NEWALLOC (rop, size);
  /* Clear last limb to prevent the case in which size is one too much.  */
  rp[size - 1] = 0;

  count = MAX_URANDOMM_ITER;	/* Set iteration count limit.  */
  do
    {
      _gmp_rand (rp, rstate, nbits);
      MPN_CMP (cmp, rp, np, size);
    }
  while (cmp >= 0 && --count != 0);

  if (count == 0)
    /* Too many iterations; return result mod n == result - n */
    mpn_sub_n (rp, rp, np, size);

  MPN_NORMALIZE (rp, size);
  SIZ (rop) = size;
  TMP_FREE;
}
```

```C
/* GMP/mini-gmp/mini-gmp.c */
void
mpz_add (mpz_t r, const mpz_t a, const mpz_t b)
{
  mp_size_t rn;

  if ( (a->_mp_size ^ b->_mp_size) >= 0)
    rn = mpz_abs_add (r, a, b);
  else
    rn = mpz_abs_sub (r, a, b);

  r->_mp_size = a->_mp_size >= 0 ? rn : - rn;
}

/* MPZ assignment and basic conversions. */
void
mpz_set_si (mpz_t r, signed long int x)
{
  if (x >= 0)
    mpz_set_ui (r, x);
  else /* (x < 0) */
    if (GMP_LIMB_BITS < GMP_ULONG_BITS)
      {
	mpz_set_ui (r, GMP_NEG_CAST (unsigned long int, x));
	mpz_neg (r, r);
      }
  else
    {
      r->_mp_size = -1;
      MPZ_REALLOC (r, 1)[0] = GMP_NEG_CAST (unsigned long int, x);
    }
}
```

## 一些特殊情况

```python
# sage
set_random_seed(seed)
a = getrandbits(nbits)

set_random_seed(seed)
b = random_matrix(Zmod(2**nbits-1), 1, 1)[0,0]

assert a == b
```

按照上述的溯源得到的底层调用

`getrandbits(nbits)` -> `random.Random().getrandbits(nbits)`

返回 nbits 位的随机数

`random_matrix(Zmod(2**nbits-1))` -> `Zmod(2**nbits-1).random_element()` -> `random.Random().randint(0, 2**nbits-2)` 等价于 `random.Random().randrange(0, 2**nbits-1)`

同样是返回 nbits 位的随机数，同时该随机数小于 `2**nbits-1`

因此在大部分的情况下，两者输出是相等的。

反倒是如果计算 `random.Random().randrange(0, 2**nbits)` 会因为 `random._randbelow()` 没有对 n 是 2 的幂次数做特判，便会计算 nbits+1 位的随机数用于拒绝采样，从而导致 和 `getrandbits(nbits)` 结果不同
